import unittest
from demo.slug import slugify_title

class SlugTest(unittest.TestCase):
    def test_whitespace(self): self.assertEqual(slugify_title("  Hello   WORLD "), "hello-world")
    def test_unicode(self): self.assertEqual(slugify_title("Straße"), "strasse")
    def test_empty(self):
        with self.assertRaises(ValueError): slugify_title("   ")
    def test_type(self):
        with self.assertRaises(TypeError): slugify_title(None)
